\section{Guide}

\begin{frame}{Reading guide}
\small
\begin{itemize}
  \item \textbf{Colors}: treatment group versus all other couples.
  \item \textbf{Columns}: wife / husband outcomes.
  \item \textbf{Rows}: couples with youngest child age 0--10 vs.\ 11--17 at baseline.
  \item \textbf{COVID figures}: COVID-study waves.
  \item \textbf{History + future figures}: regular main-study waves before and after 2020.
\end{itemize}

\vspace{0.4em}
\textbf{Treatments shown in the deck}
\begin{itemize}
  \item \textbf{Main}: wife key worker (non-education), husband not key worker or education only.
  \item \textbf{Second}: husband in shutdown sector, wife not.
\end{itemize}
\end{frame}

\begin{frame}{Event-study methodology}
\small
Event studies are estimated separately by study sample, treatment definition, spouse, outcome, and baseline child-age group.

\vspace{0.4em}
\[
Y_{ict}
=
\alpha_t
+
\delta T_c
+
\sum_{\tau \neq \tau_0}
\beta_{\tau}
\left[
\mathbf{1}\{t=\tau\}\times T_c
\right]
+
X_c'\gamma
+
\varepsilon_{ict}.
\]

\vspace{0.2em}
\scriptsize
\begin{itemize}
  \item \(T_c\): couple-level treatment indicator.
  \item \(\alpha_t\): period fixed effects.
  \item \(\tau_0\): omitted reference period.
  \item Main-study event studies use 2019 as the reference year.
  \item COVID-study event studies use 2019 where available; WFH/work-outside use Jan--Feb 2020.
  \item Figures use no couple fixed effects.
  \item Standard errors are clustered at the person level.
  \item Baseline covariates: wife and husband age, wife and husband education category, number of children under 18, number of children under 10, and region.
\end{itemize}

\end{frame}

\section{Appendix: outcome variable definitions}

\begin{frame}{Outcome variable definitions}
\centering
\tiny
\begin{adjustbox}{max width=\linewidth,max totalheight=0.78\textheight,center}
\begin{tabularx}{\linewidth}{@{}p{0.22\linewidth}X@{}}
\toprule
\textbf{Variable} & \textbf{Definition used in the couple-treatment outputs} \\
\midrule
\texttt{any\_work} & Indicator for worked positive hours. In COVID waves this uses hours worked last week; in main/future waves this uses positive usual weekly hours among employed or self-employed respondents. \\
\texttt{hours} & COVID-study hours worked last week. Synthetic pre-COVID rows use Jan--Feb 2020 baseline hours where available. \\
\texttt{jbhrs} & Main/future usual weekly hours worked; event-study cleaning sets this to zero for non-workers while leaving missing values among workers missing. \\
\texttt{workoutside} & Indicator for working outside the home. Equals one for workers who do not always work from home; equals zero for non-workers, zero hours, or always working from home. \\
\texttt{wfh\_some} & Indicator for working from home always, often, or sometimes. The figure stem uses \texttt{wfh\_any}. Defined only in waves where a WFH question is available. \\
\texttt{howlng} & Weekly housework hours. COVID raw \texttt{howlng\_cv} is harmonized to \texttt{howlng}; the Jan--Feb 2020 synthetic baseline row has no comparable housework measure. \\
\texttt{timechcare} & COVID-study childcare / home-schooling hours; synthetic 2019 and Jan--Feb 2020 rows are missing. \\
\texttt{paygu\_dv} & Usual gross pay per month from current job; main event studies set this to zero for non-workers. \\
\texttt{fimnlabgrs\_dv} & Gross monthly labour income; main event studies set this to zero for non-workers. \\
\texttt{fimngrs\_dv} & Gross monthly personal income; main event studies set this to zero for non-workers. \\
\texttt{husits\_wife\_main\_both} & Indicator equal to one when both spouses' harmonized childcare-responsibility reports imply the wife is mainly responsible. \\
\bottomrule
\end{tabularx}
\end{adjustbox}

\end{frame}
