\section{Guide}

\begin{frame}{Reading guide}
\small
\begin{itemize}
  \item \textbf{Colors}: treatment group versus all other couples.
  \item \textbf{Columns}: wife / husband outcomes.
  \item \textbf{Rows}: couples with youngest child age 0--10 vs.\ 11--17 at baseline.
  \item \textbf{COVID figures}: COVID-study waves.
  \item \textbf{Main-study figures}: regular UKHLS main-study waves before and after 2020, limited to couples observed in the COVID sample.
  \item \textbf{Pay outcomes}: shown in constant December 2019 pounds.
  \item \textbf{Appendix}: full-sample main-study versions and covariate event-study figures.
\end{itemize}

\vspace{0.4em}
\textbf{Treatments shown in the deck}
\begin{itemize}
  \item \textbf{Main}: wife key worker (non-education), husband not key worker or education only.
  \item \textbf{Second}: husband in shutdown sector, wife not.
\end{itemize}
\end{frame}

\begin{frame}{Event-study methodology}
\small
Event studies are estimated separately by study sample, treatment definition, spouse, outcome, and baseline child-age group.

\vspace{0.4em}
\[
Y_{ict}
=
\alpha_t
+
\delta T_c
+
\sum_{\tau \neq \tau_0}
\beta_{\tau}
\left[
\mathbf{1}\{t=\tau\}\times T_c
\right]
+
X_c'\gamma
+
\varepsilon_{ict}.
\]

\vspace{0.2em}
\scriptsize
\begin{itemize}
  \item \(T_c\): couple-level treatment indicator.
  \item \(\alpha_t\): period fixed effects.
  \item \(\tau_0\): omitted reference period.
  \item Main-study event studies use 2019 as the reference year.
  \item COVID-study event studies use 2019 where available; WFH/work-outside use Jan--Feb 2020.
  \item Figures use no couple fixed effects.
  \item Standard errors are clustered at the person level.
  \item Baseline covariates: wife and husband age, wife and husband education category, number of children under 18, number of children under 10, and region.
\end{itemize}

\end{frame}

\section{Appendix: real-pay adjustment}

\begin{frame}{Appendix: real-pay adjustment}
\small
All pay and income outcomes in this deck are shown in constant December 2019 pounds.

\vspace{0.6em}
\[
Y^{real}_{it}
=
Y^{nominal}_{it}
\times
\frac{\text{CPIH}_{2019m12}}{\text{CPIH}_{t}}.
\]

\vspace{0.4em}
\begin{itemize}
  \item Price index: ONS CPIH all-items index, monthly series \texttt{L522}, \emph{CPIH INDEX 00: ALL ITEMS 2015=100}.
  \item Base month: December 2019; \(\text{CPIH}_{2019m12}=108.5\).
  \item UKHLS negative monetary codes are treated as missing before deflation.
  \item Nominal variables are retained in the data; analysis-facing pay figures and event studies use the \texttt{\_real} versions.
\end{itemize}

\vspace{0.4em}
\scriptsize
Source: Office for National Statistics, CPIH all-items monthly time series \texttt{L522}: \url{https://www.ons.gov.uk/economy/inflationandpriceindices/timeseries/l522/mm23}
\end{frame}

\section{Appendix: outcome variable definitions}

\begin{frame}{Outcome variable definitions}
\centering
\tiny
\begin{adjustbox}{max width=\linewidth,max totalheight=0.78\textheight,center}
\begin{tabularx}{\linewidth}{@{}p{0.22\linewidth}X@{}}
\toprule
\textbf{Variable} & \textbf{Definition used in the couple-treatment outputs} \\
\midrule
\texttt{any\_work} & Indicator for worked positive hours. In COVID waves this uses hours worked last week; in main-study waves this uses positive usual weekly hours among employed or self-employed respondents. \\
\texttt{hours} & COVID-study hours worked last week. Synthetic pre-COVID rows use Jan--Feb 2020 baseline hours where available. \\
\texttt{jbhrs} & Main-study usual weekly hours worked; event-study cleaning sets this to zero for non-workers while leaving missing values among workers missing. \\
\texttt{workoutside} & Indicator for working outside the home. Equals one for workers who do not always work from home; equals zero for non-workers, zero hours, or always working from home. \\
\texttt{wfh\_some} & Indicator for working from home always, often, or sometimes. The figure stem uses \texttt{wfh\_any}. Defined only in waves where a WFH question is available. \\
\texttt{paygu\_dv\_real} & Usual gross pay per month from current job, converted to December 2019 pounds; main event studies set this to zero for non-workers. \\
\texttt{fimnlabgrs\_dv\_real} & Gross monthly labour income, converted to December 2019 pounds; main event studies set this to zero for non-workers. \\
\texttt{howlng} & Weekly housework hours. COVID raw \texttt{howlng\_cv} is harmonized to \texttt{howlng}; the Jan--Feb 2020 synthetic baseline row has no comparable housework measure. \\
\texttt{timechcare} & COVID-study childcare / home-schooling hours. It has no reference period for event studies, so it appears as a COVID descriptive outcome only. \\
\texttt{husits\_wife\_main\_both} & Indicator equal to one when both spouses' harmonized childcare-responsibility reports imply the wife is mainly responsible. \\
\bottomrule
\end{tabularx}
\end{adjustbox}

\vspace{0.2em}
\tiny \emph{Interpretation note:} COVID-study work outcomes are short-run, last-week measures; regular/main-study outcomes are usual-week measures. Thus \texttt{hours} and \texttt{jbhrs} are comparable as weekly-hours outcomes but not identical questionnaire concepts.

\vspace{0.2em}
\tiny Sources: code documentation in \texttt{work\_groups.R}, \texttt{covid\_loader.R}, \texttt{covid\_panel.R}, \texttt{future\_outcomes.R}, \texttt{couple\_plot\_helpers.R}, and \texttt{event\_study\_regressions.R}.
\end{frame}
