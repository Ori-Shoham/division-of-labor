\section{Guide}

\begin{frame}{Reading guide}
\small
\begin{itemize}
  \item \textbf{Colors}: treatment group versus all other couples.
  \item \textbf{Columns}: wife / husband outcomes.
  \item \textbf{Rows}: couples with youngest child age 0--10 vs.\ 11--17 at baseline.
  \item \textbf{COVID figures}: COVID-study waves.
  \item \textbf{Main-study figures}: regular UKHLS main-study waves before and after 2020, limited to couples observed in the COVID sample.
  \item \textbf{Pay outcomes}: shown in constant December 2019 pounds.
  \item \textbf{Definition slides}: each outcome opens with a slide defining the variable, its UKHLS source, reference period, and processing.
  \item \textbf{Distribution slides}: 2019 baseline (childcare hours use COVID wave 1), couples with children, full sample (no treatment split).
  \item \textbf{Appendix}: full-sample main-study versions and event studies.
\end{itemize}

\vspace{0.4em}
\textbf{Treatment shown in this deck}
\begin{itemize}
  \item Wife key worker (non-education), husband not key worker or education only.
\end{itemize}
\end{frame}

\begin{frame}{Event-study methodology}
\small
Event studies are estimated separately by study sample, treatment definition, spouse, outcome, and baseline child-age group.

\vspace{0.4em}
\[
Y_{ict}
=
\alpha_t
+
\delta T_c
+
\sum_{\tau \neq \tau_0}
\beta_{\tau}
\left[
\mathbf{1}\{t=\tau\}\times T_c
\right]
+
X_c'\gamma
+
\varepsilon_{ict}.
\]

\vspace{0.2em}
\scriptsize
\begin{itemize}
  \item \(T_c\): couple-level treatment indicator.
  \item \(\alpha_t\): period fixed effects.
  \item \(\tau_0\): omitted reference period.
  \item Main-study event studies use 2019 as the reference year.
  \item COVID-study event studies use 2019 where available; WFH/work-outside use Jan--Feb 2020.
  \item Figures use no couple fixed effects.
  \item Standard errors are clustered at the person level.
  \item Baseline covariates: wife and husband age, wife and husband education category, number of children under 18, number of children under 10, and region.
\end{itemize}

\end{frame}

\section{Appendix: real-pay adjustment}

\begin{frame}{Appendix: real-pay adjustment}
\small
All pay and income outcomes in this deck are shown in constant December 2019 pounds.

\vspace{0.6em}
\[
Y^{real}_{it}
=
Y^{nominal}_{it}
\times
\frac{\text{CPIH}_{2019m12}}{\text{CPIH}_{t}}.
\]

\vspace{0.4em}
\begin{itemize}
  \item Price index: ONS CPIH all-items index, monthly series \texttt{L522}, \emph{CPIH INDEX 00: ALL ITEMS 2015=100}.
  \item Base month: December 2019; \(\text{CPIH}_{2019m12}=108.5\).
  \item UKHLS negative monetary codes are treated as missing before deflation.
  \item Nominal variables are retained in the data; analysis-facing pay figures and event studies use the \texttt{\_real} versions.
\end{itemize}

\vspace{0.4em}
\scriptsize
Source: Office for National Statistics, CPIH all-items monthly time series \texttt{L522}: \url{https://www.ons.gov.uk/economy/inflationandpriceindices/timeseries/l522/mm23}
\end{frame}
